%%% Generating sin waves with diffrent frequencies and ampitudes
clc
clear
close all
fs = 500; % define a sampling rate


frq = [ 5   7   15   26   30 ]; % list some frequencies
Amp = [ 2   15   10   5   7 ];
phases = [  pi/7  pi/8  pi  pi/2  -pi/4 ]; % phases... list some random numbers between -pi and pi

TimeVec = 0:1/fs:1; % define time...

% now loop through frequencies and create sine waves, created matrix of
% zeros
sine_waves = zeros(length(frq), length(TimeVec));

%disp(sine_waves)


Colors = {'b','g','r','m','c'};
for fi=1:length(frq)
    sine_waves(fi,:) = Amp(fi) * sin(2*pi*TimeVec*frq(fi) + phases(fi));

end
%%% plot
figure, clf

for i=1:length(Amp)
    subplot(length(frq),1,i)
    plot(TimeVec,sine_waves(i,:),'color',Colors{i},'linew',1.5)
    title(['Frequeny:', num2str(frq(i)) ,', Amplitude: ', num2str(Amp(i)) ' ,Phase:', num2str(phases(i))])
end

Timeseries = sum(sine_waves,1);
figure, clf
subplot(3,1,1)

plot(TimeVec,Timeseries,'color','k','linew',1)
xlabel('Time (s)')
title('Original Time series ')


littleNoise = 10*randn(1,length(Timeseries));       % generate random noise with low amplitude
lotsOfNoise = 100*randn(1,length(Timeseries));    % generate random noise with high amplitude
% Adding noise to original signals
TimeseriesWithLowNoise = Timeseries + littleNoise; 
TimeseriesWithHighNoise = Timeseries + lotsOfNoise;


% plot summed sine waves with little noise
subplot(3,1,2)
plot(TimeVec,TimeseriesWithLowNoise,'k')
title('Time series with LITTLE noise')

% plot summed sine waves with high noise

subplot(3,1,3)
plot(TimeVec,TimeseriesWithHighNoise,'k')
title('Time series with A LOT of noise')
