clc
clear

% Create two vectors
a = [2 4 2 1 5 3 1];
b = [4 2 2 -3 2 5 0];

% Dot product (two equivalent ways)
DotPro1 = sum(a .* b) % Manual dot product
DotPro2 = dot(a, b) % Matlab built in function - same result

% Time axis, signal is vector
fs = 1000; %Sampling frequency
t = -1:1/fs:1; %Time vector

% Gaussian window (keeps signal localized and smooth bell-shaped)
delta = 0.1;
gauss = exp(-(t.^2)/delta);

% Signal: sine wave multiplied by Gaussian
f0 = 5;          % signal frequency (Hz)
theta = pi;      % phase
signal = sin(2*pi*f0*t + theta) .*gauss;

% Plot signal
figure(1), clf
subplot(2,1,1)
plot(t, signal, 'k', 'LineWidth', 2)
xlabel('Time (sec)')
ylabel('Amplitude')
title('Signal')

% Frequencies to compare
frequencies = 1:10;
DotPro = zeros(size(frequencies));

for i = 1:length(frequencies)
    testSine = sin(2*pi*frequencies(i)*t);
    DotPro(i) = dot(testSine, signal) / length(t);
end

% Plot dot product results
subplot(2,1,2)
stem(frequencies, DotPro, 'filled') %to create discrete plot
xlabel('Sine wave frequency (Hz)')
ylabel('Dot product')
title('Dot product between signal and sine waves')
